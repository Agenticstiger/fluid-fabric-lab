# manifests are expected to need. Add a kind here, then run ../regenerate.sh.

# yannh/kubernetes-json-schema (the repo omits apiextensions.k8s.io/v1
# CustomResourceDefinition); regenerate.sh records it in MISSING.txt and the
# schema check must treat it as an explicit, reviewed exception.
